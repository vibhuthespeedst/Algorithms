#include <bits/stdc++.h>
using namespace std;

int dfs(int node, vector<vector<int>>&adj, int par, int& res){
    int first = 0, second = 0;
    for(int &child:adj[node]){
        if(child==par) continue;
        int depth = 1 + dfs(child, adj, node, res);

        if (depth > first) {
            second = first;
            first = depth;
        } else if (depth > second) {
            second = depth;
        }
    }
    res = max(res, first + second);
    return first; 
}

int main() {
    int n;
    cin >> n;
    vector<vector<int>> adj(n + 1);
    for (int i = 1; i < n; i++) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    int res = 0;
    dfs(1,adj,0,res);

    cout << res <<endl;
}

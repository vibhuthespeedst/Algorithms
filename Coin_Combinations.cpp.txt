#include <bits/stdc++.h>
using namespace std;

int MOD = 1e9+7;

int main() {
	// your code goes here
	int n;
	cin>>n;
	vector<int> dp(max(n+1,7));
	dp[1] = 1;
	dp[2] = 2;
	dp[3] = 4;
	dp[4] = 8;
	dp[5] = 16;
	dp[6] = 32;
	for(int i=7;i<=n;i++){
	    for(int j=max(1,i-6);j<i;j++){
	        dp[i]=(dp[i]+dp[j])%MOD;   
	    }
	}
    cout<<dp[n]<<endl;
}

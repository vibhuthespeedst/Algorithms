#include <bits/stdc++.h>
using namespace std;
#define ll long long

int main() {
	// your code goes here
	ll n,x;
	cin>>n>>x;
	ll arr[n];
	for(ll i=0;i<n;i++){
	    cin>>arr[i];
	}
	sort(arr,arr+n);
	vector<ll> dp(x+1,LLONG_MAX);
	dp[0] = 0;
	for(ll i=1;i<=x;i++){
	    for(auto &it:arr){
	        if(it>i) break;
	        if (dp[i - it] != LLONG_MAX) dp[i] = min(dp[i],1+dp[i-it]);
	    } 
	}
	ll ans = (dp[x]==LLONG_MAX)?-1:dp[x];
	cout<<ans<<endl;
	return 0;
}
